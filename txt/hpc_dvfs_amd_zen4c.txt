SHORT HPC_DVFS_MODEL_ZEN4
EVENTSET
FIXC1 ACTUAL_CPU_CLOCK
FIXC2 MAX_CPU_CLOCK
PMC0  RETIRED_INSTRUCTIONS
PMC1  RETIRED_SSE_AVX_FLOPS_ALL
PMC2  RETIRED_FP_OPS_BY_TYPE_SCALAR_ALL
PMC3  DEMAND_DATA_CACHE_FILLS_ALL       # Memory pressure signal
PMC4  RETIRED_MISP_BRANCH_INSTR
PMC5  DISPATCH_STALLS_PER_SLOT_BACKEND

METRICS
# Efficiency baseline
CPI FIXC1/PMC0

# How much work is actually getting done?
Compute_Density (PMC1+PMC2)/FIXC1

# Memory wall pressure
Mem_Boundness PMC3/PMC0

# The signal to lower frequency (Wait Ratio)
# Zen 4 has 6 dispatch slots per cycle. 
# We divide by (6 * FIXC1) to get a 0.0 to 1.0 ratio.
Stall_Ratio PMC5/(6*FIXC1)

# Branch Penalty
Branch_MPKI 1000*PMC4/PMC0

# Is it heavy AVX-512? (Critical for Zen 4 thermal/power management)
Vector_Intensity PMC1/(PMC1+PMC2+1e-9)

# Current Frequency State
Clock_Ratio FIXC1/FIXC2
